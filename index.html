function renderPlans() { const selectedDate = $("plan-date").value; const selectedLine = $("plan-line").value; const filtered = plans.filter(row => (!selectedDate || row.date === selectedDate) && (!selectedLine || row.line === selectedLine) ); const dates = [...new Set( filtered.map(row => row.date) )].sort(); const models = [...new Set( filtered.map(row => row.model) )].sort((a, b) => a.localeCompare(b)); // รวม Order ตาม MODEL และวันที่ const orders = new Map(); const dateTotals = new Map(); filtered.forEach(row => { if (!orders.has(row.model)) { orders.set(row.model, new Map()); } const modelOrders = orders.get(row.model); modelOrders.set( row.date, (modelOrders.get(row.date) || 0) + row.quantity ); dateTotals.set( row.date, (dateTotals.get(row.date) || 0) + row.quantity ); }); function displayDate(date) { const [year, month, day] = date.split("-"); return `${day}/${month}/${year}`; } const table = $("plan-body").closest("table"); table.querySelector("thead").innerHTML = ` MODEL ${dates.map(date => ` ${displayDate(date)}
Order
`).join("")} `; if (!filtered.length) { $("plan-body").innerHTML = ` ยังไม่มีแผนผลิตที่ตรงกับตัวกรอง `; } else { $("plan-body").innerHTML = models.map(model => { const modelOrders = orders.get(model); return ` ${escapeHtml(model)} ${dates.map(date => ` ${modelOrders.has(date) ? format(modelOrders.get(date)) : "—"} `).join("")} `; }).join("") + ` Total Order ${dates.map(date => ` ${format(dateTotals.get(date) || 0)} `).join("")} `; } $("plan-total").textContent = format( filtered.reduce((sum, row) => sum + row.quantity, 0) ); $("plan-count").textContent = format(filtered.length); }
